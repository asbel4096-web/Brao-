// ⚠️ هذا snippet لإضافته إلى firestore.rules الموجود.
// أضيفيه قبل القوس الختامي لـmatch /databases/{database}/documents.

// ============================================================
// dealerStories - قصص المعارض الموثقة
// ============================================================
//   {dealerUid, category, mediaURL, mediaType, caption, listingId, createdAt, expiresAt}
//
// القراءة: مفتوحة (الجميع يرى القصص)
//
// الإنشاء: فقط المعارض الموثقة، dealerUid == auth.uid
//   - يجب أن يكون category صحيحاً
//   - mediaURL مطلوب
//   - dealerUid مطابق للمستخدم
//
// التحديث/الحذف: المالك فقط (يحذف قصته)
//   - الأدمن يستطيع الحذف لأي قصة (moderation)

match /dealerStories/{storyId} {
  allow read: if true;

  allow create: if signedIn()
    && request.resource.data.dealerUid == request.auth.uid
    && request.resource.data.mediaURL is string
    && request.resource.data.mediaURL.size() > 0
    && request.resource.data.category in [
      'new_arrivals', 'offers', 'showroom', 'test_drive'
    ]
    && (
      // فقط الموثقون: لديهم verifiedUntil > now أو isVerifiedDealer
      get(/databases/$(database)/documents/users/$(request.auth.uid))
        .data.get('verifiedUntil', null) != null
      || get(/databases/$(database)/documents/users/$(request.auth.uid))
        .data.get('isVerifiedDealer', false) == true
    );

  allow update: if signedIn()
    && resource.data.dealerUid == request.auth.uid
    // فقط الـviewCount يمكن تعديله من client (لـanalytics)
    && request.resource.data.diff(resource.data).affectedKeys()
       .hasOnly(['viewCount']);

  allow delete: if isAdmin()
    || (signedIn() && resource.data.dealerUid == request.auth.uid);
}
