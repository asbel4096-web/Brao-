// أضف هذا الـ block قبل قاعدة "match /{document=**}" الأخيرة في firestore.rules
// ============================================================================

// =======================================================================
// support_messages/{id}
// =======================================================================
// رسائل الدعم/الاقتراحات من المستخدمين.
// - إنشاء: أي زائر (مسجَّل أو لا) لأن قد يُرسلها قبل التسجيل
// - قراءة: الأدمن فقط
// - تحديث/حذف: الأدمن فقط
// =======================================================================
match /support_messages/{messageId} {
  allow read: if isAdmin();

  allow create: if
    // يجب وجود حقل message + name + contact
    request.resource.data.message is string
    && request.resource.data.message.size() > 0
    && request.resource.data.message.size() <= 1000
    && request.resource.data.name is string
    && request.resource.data.contact is string
    && request.resource.data.type in ['suggestion', 'issue', 'other']
    && request.resource.data.status == 'open'
    // لو مسجَّل، userId يجب أن يطابق
    && (
      !('userId' in request.resource.data)
      || request.resource.data.userId == null
      || request.resource.data.userId == request.auth.uid
    );

  allow update, delete: if isAdmin();
}
